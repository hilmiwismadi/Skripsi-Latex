\subsection{Dense Embedding (\texttt{dense/embedding\_model.py})}

Fungsi-fungsi berikut menangani konversi teks kontrol ISO menjadi representasi vektor semantik yang digunakan pada tahap Basis Pengetahuan dan Pemrosesan \textit{Query}.

\begin{lstlisting}[language=Python, caption={Fungsi konversi kontrol ISO ke teks, encoding, dan normalisasi vektor}, label={lst:embedding_model}]
import numpy as np
from sentence_transformers import SentenceTransformer

MODEL_NAME = "paraphrase-multilingual-MiniLM-L12-v2"
_model = None

def get_model() -> SentenceTransformer:
    global _model
    if _model is None:
        _model = SentenceTransformer(MODEL_NAME)
    return _model

def normalize(vectors: np.ndarray) -> np.ndarray:
    norms = np.linalg.norm(vectors, axis=1, keepdims=True)
    norms = np.where(norms == 0, 1.0, norms)
    return vectors / norms

def encode_texts(texts: list[str]) -> np.ndarray:
    model = get_model()
    embeddings = model.encode(texts, convert_to_numpy=True,
                              show_progress_bar=False)
    embeddings = embeddings.astype(np.float32)
    return normalize(embeddings)

def control_to_text(control: dict) -> str:
    parts = [
        control.get('title', ''),
        control.get('objective', ''),
        control.get('description', ''),
    ]
    keywords_en = control.get('keywords_en', [])
    keywords_id = control.get('keywords_id', [])
    if keywords_en or keywords_id:
        parts.append(' '.join(keywords_en))
        parts.append(' '.join(keywords_id))
    audit_indicators_en = control.get('audit_indicators_en', [])
    audit_indicators_id = control.get('audit_indicators_id', [])
    if audit_indicators_en or audit_indicators_id:
        parts.append(' '.join(audit_indicators_en))
        parts.append(' '.join(audit_indicators_id))
    related_assets_en = control.get('related_assets_en', [])
    related_assets_id = control.get('related_assets_id', [])
    if related_assets_en or related_assets_id:
        parts.append(' '.join(related_assets_en))
        parts.append(' '.join(related_assets_id))
    security_principles_en = control.get('security_principles_en', [])
    security_principles_id = control.get('security_principles_id', [])
    if security_principles_en or security_principles_id:
        parts.append(' '.join(security_principles_en))
        parts.append(' '.join(security_principles_id))
    return ' | '.join(parts)
\end{lstlisting}

\subsection{Sparse Retrieval -- BM25 (\texttt{sparse/bm25.py})}

Kelas berikut menangani pembangunan indeks BM25 dan penghitungan skor kecocokan kata kunci antara \textit{query} dan dokumen kontrol ISO.

\begin{lstlisting}[language=Python, caption={Implementasi BM25: tokenisasi, indeks, dan scoring}, label={lst:bm25}]
import math
import re
from collections import Counter

def tokenize(text: str) -> list[str]:
    return re.findall(r"[A-Za-z0-9]+", text.lower())

class BM25Retriever:
    def __init__(self, documents: list[str],
                 k1: float = 1.5, b: float = 0.75):
        self.k1 = k1
        self.b = b
        self.documents = documents
        self.tokenized_docs = [tokenize(doc) for doc in documents]
        self.doc_lens = [len(doc) for doc in self.tokenized_docs]
        self.avg_doc_len = sum(self.doc_lens) / max(len(self.doc_lens), 1)
        self.doc_freq = self._compute_doc_freq()
        self.idf = self._compute_idf()
        self.term_freqs = [Counter(doc) for doc in self.tokenized_docs]

    def _compute_doc_freq(self) -> dict[str, int]:
        doc_freq: dict[str, int] = {}
        for doc in self.tokenized_docs:
            unique_terms = set(doc)
            for term in unique_terms:
                doc_freq[term] = doc_freq.get(term, 0) + 1
        return doc_freq

    def _compute_idf(self) -> dict[str, float]:
        n_docs = len(self.tokenized_docs)
        idf: dict[str, float] = {}
        for term, freq in self.doc_freq.items():
            idf[term] = math.log(1 + (n_docs - freq + 0.5) / (freq + 0.5))
        return idf

    def score_query(self, query: str) -> list[float]:
        query_terms = tokenize(query)
        scores: list[float] = []
        for idx, tf_doc in enumerate(self.term_freqs):
            score = 0.0
            doc_len = self.doc_lens[idx]
            denom_norm = self.k1 * (1 - self.b + self.b
                          * doc_len / max(self.avg_doc_len, 1e-9))
            for term in query_terms:
                if term not in tf_doc:
                    continue
                tf = tf_doc[term]
                term_idf = self.idf.get(term, 0.0)
                numerator = tf * (self.k1 + 1)
                denominator = tf + denom_norm
                score += term_idf * (numerator / max(denominator, 1e-9))
            scores.append(score)
        return scores
\end{lstlisting}

\subsubsection{Algoritma Okapi BM25}

Algoritma berikut merupakan pseudocode dari metode Okapi BM25 yang diimplementasikan pada \texttt{sparse/bm25.py}. Rumus IDF yang digunakan mengadopsi varian $\log(1 + x)$ untuk menghindari nilai IDF negatif.

\begin{algorithm}[H]
\caption{Okapi BM25 Scoring}
\label{alg:bm25}
\begin{algorithmic}[1]
\Require Korpus $D = \{d_1, d_2, \ldots, d_N\}$, \textit{query} $q$, parameter $k_1 = 1{,}5$, $b = 0{,}75$
\Ensure Daftar skor BM25 $\mathit{score}(d_i)$ untuk setiap dokumen $d_i$
\State $\mathit{avgdl} \gets \frac{1}{N} \sum_{i=1}^{N} |d_i|$ \Comment{Rata-rata panjang dokumen}
\For{setiap \textit{term} $t$ dalam korpus}
    \State $\mathit{df}(t) \gets |\{d_i \in D : t \in d_i\}|$ \Comment{Document frequency}
    \State $\mathit{IDF}(t) \gets \log\left(1 + \frac{N - \mathit{df}(t) + 0{,}5}{\mathit{df}(t) + 0{,}5}\right)$
\EndFor
\For{setiap dokumen $d_i$ dalam korpus}
    \State $\mathit{score}(d_i) \gets 0$
    \For{setiap \textit{term} $t$ dalam $q$}
        \If{$t \in d_i$}
            \State $\mathit{tf}(t, d_i) \gets$ frekuensi kemunculan $t$ pada $d_i$
            \State $\mathit{norm}_i \gets k_1 \cdot \left(1 - b + b \cdot \frac{|d_i|}{\mathit{avgdl}}\right)$
            \State $\mathit{score}(d_i) \gets \mathit{score}(d_i) + \mathit{IDF}(t) \cdot \frac{\mathit{tf}(t, d_i) \cdot (k_1 + 1)}{\mathit{tf}(t, d_i) + \mathit{norm}_i}$
        \EndIf
    \EndFor
\EndFor
\State \Return $\mathit{score}(d_1), \mathit{score}(d_2), \ldots, \mathit{score}(d_N)$
\end{algorithmic}
\end{algorithm}

\subsection{Hybrid Retriever (\texttt{hybrid/hybrid\_retriever.py})}

Kelas berikut mengorkestrasi penggabungan skor \textit{dense} dan \textit{sparse} menjadi \textit{hybrid score}, sesuai tahap Pemrosesan \textit{Scoring} pada Subbab~3.2.4.1.

\begin{lstlisting}[language=Python, caption={Hybrid retrieval: normalisasi, fusi skor, dan pemilihan dokumen teratas}, label={lst:hybrid_retriever}]
import json
import numpy as np
from pathlib import Path
from dense.embedding_model import control_to_text, encode_texts
from sparse.bm25 import BM25Retriever

class HybridRetriever:
    def __init__(self, controls_path: Path):
        self.controls_path = controls_path
        self.controls = self._load_controls()
        self.control_texts = [control_to_text(c) for c in self.controls]
        self.doc_embeddings = encode_texts(self.control_texts)
        self.bm25 = BM25Retriever(self.control_texts)

    def _load_controls(self) -> list[dict]:
        with self.controls_path.open("r", encoding="utf-8") as f:
            return json.load(f)

    @staticmethod
    def _minmax(scores: list[float]) -> np.ndarray:
        arr = np.array(scores, dtype=np.float32)
        min_v, max_v = float(np.min(arr)), float(np.max(arr))
        if max_v - min_v <= 1e-12:
            return np.zeros_like(arr)
        return (arr - min_v) / (max_v - min_v)

    def _dense_scores(self, query: str) -> np.ndarray:
        q_emb = encode_texts([query])[0]
        return np.dot(self.doc_embeddings, q_emb)

    def _build_result(self, idx, dense_score, sparse_score,
                      hybrid_score) -> dict:
        control = self.controls[idx]
        return {
            "control_id": control["control_id"],
            "title": control["title"],
            "objective": control["objective"],
            "description": control["description"],
            "dense_score": float(dense_score),
            "sparse_score": float(sparse_score),
            "hybrid_score": float(hybrid_score),
        }

    def retrieve(self, query: str, k: int = 3,
                 dense_weight: float = 0.6,
                 sparse_weight: float = 0.4) -> dict:
        _dw = dense_weight
        _sw = sparse_weight

        dense_scores = self._dense_scores(query)
        sparse_scores = np.array(
            self.bm25.score_query(query), dtype=np.float32)
        dense_norm = self._minmax(dense_scores.tolist())
        sparse_norm = self._minmax(sparse_scores.tolist())
        hybrid_scores = _dw * dense_norm + _sw * sparse_norm

        hybrid_rank = np.argsort(-hybrid_scores)[:k]
        hybrid_results = [
            self._build_result(i,
                float(dense_scores[i]), float(sparse_scores[i]),
                float(hybrid_scores[i]))
            for i in hybrid_rank
        ]
        return {"hybrid_results": hybrid_results}
\end{lstlisting}

\subsubsection{Algoritma Hybrid Retrieval}

Algoritma berikut merupakan pseudocode dari proses fusi skor \textit{dense} dan \textit{sparse} menggunakan normalisasi \textit{min-max} dan \textit{weighted sum}, yang diimplementasikan pada \texttt{hybrid/hybrid\_retriever.py}.

\begin{algorithm}[H]
\caption{Hybrid Retrieval Score Fusion}
\label{alg:hybrid_retrieval}
\begin{algorithmic}[1]
\Require \textit{Query} $q$, matriks \textit{embedding} dokumen $\mathbf{E} \in \mathbb{R}^{N \times d}$, indeks BM25, bobot $w_{\textit{dense}}$, $w_{\textit{sparse}}$, jumlah hasil $k$
\Ensure Daftar $k$ dokumen teratas beserta skor
\State $\mathbf{q} \gets \mathit{encode}(q)$ \Comment{Encode query ke vektor 384-dim, L2-normalize}
\State $\mathit{dense\_scores} \gets \mathbf{E} \cdot \mathbf{q}$ \Comment{Dot product = cosine similarity (vektor ternormalisasi)}
\State $\mathit{sparse\_scores} \gets \mathit{BM25}(q)$ \Comment{Skor BM25 untuk seluruh dokumen}
\State $\mathit{dense\_norm}_i \gets \frac{\mathit{dense\_scores}_i - \min(\mathit{dense\_scores})}{\max(\mathit{dense\_scores}) - \min(\mathit{dense\_scores})}$ \Comment{Min-max normalisasi}
\State $\mathit{sparse\_norm}_i \gets \frac{\mathit{sparse\_scores}_i - \min(\mathit{sparse\_scores})}{\max(\mathit{sparse\_scores}) - \min(\mathit{sparse\_scores})}$
\For{$i = 1$ \textbf{to} $N$}
    \State $\mathit{hybrid\_scores}_i \gets w_{\textit{dense}} \cdot \mathit{dense\_norm}_i + w_{\textit{sparse}} \cdot \mathit{sparse\_norm}_i$
\EndFor
\State $\mathit{top\_k} \gets$ indeks $k$ dokumen dengan $\mathit{hybrid\_scores}$ tertinggi
\State \Return $\mathit{top\_k}$ beserta $\mathit{dense\_scores}$, $\mathit{sparse\_scores}$, $\mathit{hybrid\_scores}$
\end{algorithmic}
\end{algorithm}

\subsection{Cross-Encoder Reranker (\texttt{reranking/cross\_encoder\_reranker.py})}

Fungsi-fungsi berikut menangani pembentukan pasangan \textit{query}--dokumen, penghitungan skor \textit{cross-encoder}, normalisasi, dan fusi skor dengan \textit{hybrid score}, sesuai tahap \textit{Cross-Encoder Reranking} pada Subbab~3.2.4.2.

\begin{lstlisting}[language=Python, caption={Cross-encoder reranking: pembentukan teks, normalisasi, dan fusi skor}, label={lst:cross_encoder_reranker}]
from typing import Any
import numpy as np
from sentence_transformers import CrossEncoder

MODEL_NAME = "BAAI/bge-reranker-v2-m3"
_model: CrossEncoder | None = None

def get_model() -> CrossEncoder:
    global _model
    if _model is None:
        _model = CrossEncoder(MODEL_NAME)
    return _model

def _doc_to_rerank_text(doc: dict[str, Any]) -> str:
    parts: list[str] = []
    title = str(doc.get("title", "")).strip()
    if title:
        parts.append(title)
    objective = str(doc.get("objective", "")).strip()
    if objective:
        parts.append(objective)
    description = str(doc.get("description", "")).strip()
    if description:
        parts.append(description)
    keywords_en = doc.get("keywords_en", [])
    if keywords_en:
        parts.append("Keywords EN: " + " ".join(str(k) for k in keywords_en))
    keywords_id = doc.get("keywords_id", [])
    if keywords_id:
        parts.append("Kata kunci ID: " + " ".join(str(k) for k in keywords_id))
    audit_indicators = (doc.get("audit_indicators_en", [])
                        or doc.get("audit_indicators_id", []))
    if audit_indicators:
        parts.append("Audit indicators: "
                      + " ".join(str(a) for a in audit_indicators))
    return " | ".join(parts)

def _normalize(arr: np.ndarray) -> np.ndarray:
    mn, mx = float(np.min(arr)), float(np.max(arr))
    if mx - mn < 1e-12:
        return np.zeros_like(arr)
    return (arr - mn) / (mx - mn)

def rerank_documents(
    query: str,
    retrieved_docs: list[dict[str, Any]],
    top_n: int | None = None,
    alpha: float = 0.8,
) -> list[dict[str, Any]]:
    if not retrieved_docs:
        return []
    _alpha = alpha

    pairs = [(query, _doc_to_rerank_text(doc))
             for doc in retrieved_docs]
    ce_scores = np.array(
        get_model().predict(pairs, show_progress_bar=False),
        dtype=np.float32)

    hybrid_scores = np.array(
        [float(doc.get("hybrid_score", 0.0)) for doc in retrieved_docs],
        dtype=np.float32)

    ce_norm = _normalize(ce_scores)
    hybrid_norm = _normalize(hybrid_scores)
    combined = _alpha * ce_norm + (1 - _alpha) * hybrid_norm

    ranked = []
    for i, doc in enumerate(retrieved_docs):
        item = dict(doc)
        item["rerank_score"] = float(ce_scores[i])
        item["combined_score"] = float(combined[i])
        ranked.append(item)

    ranked.sort(key=lambda x: x["combined_score"], reverse=True)
    if top_n is not None:
        return ranked[:max(int(top_n), 0)]
    return ranked
\end{lstlisting}

\subsubsection{Algoritma Cross-Encoder Reranking}

Algoritma berikut merupakan pseudocode dari proses \textit{reranking} menggunakan \textit{cross-encoder} dan fusi skor dengan \textit{hybrid score}, yang diimplementasikan pada \texttt{reranking/cross\_encoder\_reranker.py}.

\begin{algorithm}[H]
\caption{Cross-Encoder Reranking with Score Fusion}
\label{alg:cross_encoder_reranking}
\begin{algorithmic}[1]
\Require \textit{Query} $q$, daftar kandidat $C = \{c_1, c_2, \ldots, c_m\}$ beserta $\mathit{hybrid\_score}_i$, bobot $\alpha$, jumlah hasil $\mathit{top\_n}$
\Ensure Daftar $\mathit{top\_n}$ dokumen terurut berdasarkan \textit{combined score}
\For{$i = 1$ \textbf{to} $m$}
    \State $\mathit{text}_i \gets \mathit{to\_rerank\_text}(c_i)$ \Comment{Konversi dokumen ke teks untuk CE}
    \State $\mathit{pair}_i \gets (q, \mathit{text}_i)$
\EndFor
\State $\mathit{ce\_scores} \gets \mathit{CrossEncoder}.\mathit{predict}(\mathit{pair}_1, \ldots, \mathit{pair}_m)$
\State $\mathit{ce\_norm}_i \gets \frac{\mathit{ce\_scores}_i - \min(\mathit{ce\_scores})}{\max(\mathit{ce\_scores}) - \min(\mathit{ce\_scores})}$ \Comment{Min-max normalisasi}
\State $\mathit{hybrid\_norm}_i \gets \frac{\mathit{hybrid\_score}_i - \min(\mathit{hybrid\_score})}{\max(\mathit{hybrid\_score}) - \min(\mathit{hybrid\_score})}$
\For{$i = 1$ \textbf{to} $m$}
    \State $\mathit{combined}_i \gets \alpha \cdot \mathit{ce\_norm}_i + (1 - \alpha) \cdot \mathit{hybrid\_norm}_i$
\EndFor
\State Urutkan $C$ berdasarkan $\mathit{combined}$ secara menurun
\State \Return $\mathit{top\_n}$ dokumen pertama beserta $\mathit{ce\_scores}$ dan $\mathit{combined}$
\end{algorithmic}
\end{algorithm}

\subsection{Perhitungan Metrik Evaluasi (\texttt{app/app.py})}

Fungsi berikut menghitung metrik evaluasi yang digunakan pada penelitian ini, yaitu Hit@$k$, Recall@$k$, MAP@$k$, dan NDCG@$k$. Fungsi ini menerima daftar \textit{control ID} hasil \textit{retrieval} dan daftar \textit{control ID} \textit{ground truth} sebagai masukan.

\begin{lstlisting}[language=Python, caption={Perhitungan metrik evaluasi: Hit, Recall, MAP, dan NDCG}, label={lst:compute_metrics}]
import math

def compute_metrics(predicted_ids: list[str],
                    ground_truth_ranked: list[str],
                    k: int = 3) -> dict[str, float | None]:
    gt = [str(x).strip() for x in ground_truth_ranked
          if str(x).strip()]
    pred = [str(x).strip() for x in predicted_ids[:k]
            if str(x).strip()]

    if not gt:
        return {"hit": None, "recall": None,
                "map": None, "ndcg": None}

    inter_count = 0
    seen: set[str] = set()
    for cid in pred:
        if cid in gt and cid not in seen:
            inter_count += 1
            seen.add(cid)

    hit = 1.0 if inter_count > 0 else 0.0
    recall = inter_count / max(len(gt), 1)

    hits = 0
    precision_sum = 0.0
    for idx, cid in enumerate(pred, start=1):
        if cid in set(gt):
            hits += 1
            precision_sum += hits / idx
    map_score = precision_sum / max(len(set(gt)), 1)

    dcg = 0.0
    for idx, cid in enumerate(pred, start=1):
        if cid in set(gt):
            dcg += 1.0 / math.log2(idx + 1)
    ideal_hits = min(len(gt), k)
    idcg = 0.0
    for rank in range(1, ideal_hits + 1):
        idcg += 1.0 / math.log2(rank + 1)
    ndcg = dcg / idcg if idcg > 0 else 0.0

    return {"hit": hit, "recall": recall,
            "map": map_score, "ndcg": ndcg}
\end{lstlisting}

\subsubsection{Algoritma Perhitungan Metrik Evaluasi}

Algoritma berikut merupakan pseudocode dari perhitungan metrik evaluasi Hit@$k$, Recall@$k$, MAP@$k$, dan NDCG@$k$ yang diimplementasikan pada \texttt{app/app.py}.

\begin{algorithm}[H]
\caption{Perhitungan Metrik Evaluasi (Hit, Recall, MAP, NDCG)}
\label{alg:compute_metrics}
\begin{algorithmic}[1]
\Require Daftar prediksi $\mathit{pred} = [p_1, p_2, \ldots, p_k]$, daftar \textit{ground truth} $\mathit{gt} = [g_1, g_2, \ldots, g_{|\mathit{gt}|}]$, parameter $k$
\Ensure $\mathit{Hit}@k$, $\mathit{Recall}@k$, $\mathit{MAP}@k$, $\mathit{NDCG}@k$
\State $\mathit{hit\_count} \gets 0$ \Comment{Jumlah item relevan yang ditemukan}
\State $\mathit{seen} \gets \emptyset$
\For{$i = 1$ \textbf{to} $k$}
    \If{$p_i \in \mathit{gt} \ \mathbf{and} \ p_i \notin \mathit{seen}$}
        \State $\mathit{hit\_count} \gets \mathit{hit\_count} + 1$
        \State $\mathit{seen} \gets \mathit{seen} \cup \{p_i\}$
    \EndIf
\EndFor
\State $\mathit{Hit}@k \gets \begin{cases} 1 & \text{jika } \mathit{hit\_count} > 0 \\ 0 & \text{jika tidak} \end{cases}$
\State $\mathit{Recall}@k \gets \dfrac{\mathit{hit\_count}}{|\mathit{gt}|}$
\State \Comment{Perhitungan MAP@$k$}
\State $\mathit{cum\_hits} \gets 0$; $\mathit{prec\_sum} \gets 0$
\For{$i = 1$ \textbf{to} $k$}
    \If{$p_i \in \mathit{gt}$}
        \State $\mathit{cum\_hits} \gets \mathit{cum\_hits} + 1$
        \State $\mathit{prec\_sum} \gets \mathit{prec\_sum} + \dfrac{\mathit{cum\_hits}}{i}$ \Comment{Precision pada posisi ke-$i$}
    \EndIf
\EndFor
\State $\mathit{MAP}@k \gets \dfrac{\mathit{prec\_sum}}{|\mathit{gt}|}$
\State \Comment{Perhitungan NDCG@$k$}
\State $\mathit{DCG} \gets 0$
\For{$i = 1$ \textbf{to} $k$}
    \If{$p_i \in \mathit{gt}$}
        \State $\mathit{DCG} \gets \mathit{DCG} + \dfrac{1}{\log_2(i + 1)}$ \Comment{Relensi biner: $\mathit{rel} = 1$}
    \EndIf
\EndFor
\State $\mathit{ideal\_hits} \gets \min(|\mathit{gt}|, k)$
\State $\mathit{IDCG} \gets 0$
\For{$r = 1$ \textbf{to} $\mathit{ideal\_hits}$}
    \State $\mathit{IDCG} \gets \mathit{IDCG} + \dfrac{1}{\log_2(r + 1)}$
\EndFor
\State $\mathit{NDCG}@k \gets \begin{cases} \dfrac{\mathit{DCG}}{\mathit{IDCG}} & \text{jika } \mathit{IDCG} > 0 \\ 0 & \text{jika tidak} \end{cases}$
\State \Return $\mathit{Hit}@k$, $\mathit{Recall}@k$, $\mathit{MAP}@k$, $\mathit{NDCG}@k$
\end{algorithmic}
\end{algorithm}
